  <script>
    const canvas = document.getElementById('wheel');
    const ctx = canvas.getContext('2d');
    const prizes = ["100 Голды", "Попробуй завтра", "250 Голды", "Премиум 1 день", "500 Голды", "Пусто"];
    const colors = ["#e67e22", "#2c3e50", "#f1c40f", "#27ae60", "#e74c3c", "#34495e"];
    
    let currentAngle = 0;

    function drawWheel() {
      const numSegments = prizes.length;
      const anglePerSegment = (2 * Math.PI) / numSegments;

      for (let i = 0; i < numSegments; i++) {
        ctx.beginPath();
        ctx.fillStyle = colors[i];
        ctx.moveTo(150, 150);
        ctx.arc(150, 150, 150, i * anglePerSegment, (i + 1) * anglePerSegment);
        ctx.fill();

        ctx.save();
        ctx.translate(150, 150);
        ctx.rotate(i * anglePerSegment + anglePerSegment / 2);
        ctx.textAlign = "right";
        ctx.fillStyle = "#fff";
        ctx.font = "bold 13px Arial";
        ctx.fillText(prizes[i], 135, 5);
        ctx.restore();
      }
    }

    drawWheel();

    function spin() {
      const spinBtn = document.getElementById('spinBtn');
      spinBtn.disabled = true;

      // Проверка на 24 часа
      const lastSpin = localStorage.getItem('lastSpin');
      const now = new Date().getTime();
      
      if (lastSpin && now - lastSpin < 86400000) {
        document.getElementById('result').innerText = "⏳ Вы уже крутили сегодня! Приходите завтра.";
        return;
      }

      const randomDegree = Math.floor(3600 + Math.random() * 360);
      currentAngle += randomDegree;
      canvas.style.transform = `rotate(${currentAngle}deg)`;

      setTimeout(() => {
        localStorage.setItem('lastSpin', new Date().getTime());
        const actualDegree = currentAngle % 360;
        const segmentAngle = 360 / prizes.length;
        const winningIndex = Math.floor((360 - (actualDegree % 360)) / segmentAngle) % prizes.length;
        
        const winText = prizes[winningIndex];
        if (winText.includes("Голды") || winText.includes("Премиум")) {
          document.getElementById('result').innerText = `🎉 Вы выиграли: ${winText}!\nНапишите админу код: #WIN_${Math.floor(Math.random()*8999+1000)}`;
        } else {
          document.getElementById('result').innerText = `😔 Приз: ${winText}. Повезет завтра!`;
        }
      }, 4000);
    }
  </script>
